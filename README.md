## 🎨 Примеры кода

### <div align="center">📝 Задание 1: Простой калькулятор</div>

```cpp
#include <iostream>
using namespace std;

int main() {
    setlocale(LC_ALL, "Russian");
    
    double num1, num2;
    char operation;
    
    cout << "=== ПРОСТОЙ КАЛЬКУЛЯТОР ===" << endl;
    cout << "Введите первое число: ";
    cin >> num1;
    
    cout << "Введите второе число: ";
    cin >> num2;
    
    cout << "Выберите операцию (+, -, *, /): ";
    cin >> operation;
    
    double result;
    bool valid = true;
    
    switch(operation) {
        case '+':
            result = num1 + num2;
            break;
        case '-':
            result = num1 - num2;
            break;
        case '*':
            result = num1 * num2;
            break;
        case '/':
            if (num2 == 0) {
                cout << "❌ Ошибка: деление на ноль!" << endl;
                valid = false;
            } else {
                result = num1 / num2;
            }
            break;
        default:
            cout << "❌ Ошибка: неизвестная операция!" << endl;
            valid = false;
    }
    
    if (valid) {
        cout << "✅ Результат: " << result << endl;
    }
    
    return 0;
}
```

**Пример работы:**
```
=== ПРОСТОЙ КАЛЬКУЛЯТОР ===
Введите первое число: 10
Введите второе число: 5
Выберите операцию (+, -, *, /): *
✅ Результат: 50
```
