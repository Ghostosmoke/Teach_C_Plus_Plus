## Вариант А: Простой (для новичков)

```cpp
#include <iostream>
using namespace std;

int main() {
    // Переменная для размера матрицы
    int n;
    
    // Спрашиваем у пользователя размер матрицы
    cout << "Введите размер матрицы: ";
    cin >> n;
    
    // Проверяем, что размер правильный
    if (n < 1 || n > 10) {
        cout << "Ошибка: размер должен быть от 1 до 10!" << endl;
        return 1;  // завершаем программу с ошибкой
    }
    
    // Создаём статический массив (фиксированный размер)
    int matrix[10][10];
    
    // Просим пользователя ввести элементы матрицы
    cout << "Введите элементы матрицы (" << n << "x" << n << "):" << endl;
    for (int i = 0; i < n; i++) {
        for (int j = 0; j < n; j++) {
            cin >> matrix[i][j];
        }
    }
    
    // Выводим матрицу на экран
    cout << "\nВаша матрица:" << endl;
    for (int i = 0; i < n; i++) {
        for (int j = 0; j < n; j++) {
            cout << matrix[i][j] << "\t";
        }
        cout << endl;
    }
    
    // Находим сумму главной диагонали
    // Главная диагональ: элементы, у которых i == j
    int mainSum = 0;
    for (int i = 0; i < n; i++) {
        mainSum += matrix[i][i];
    }
    
    // Находим сумму побочной диагонали
    // Побочная диагональ: элементы, у которых i + j == n - 1
    int sideSum = 0;
    for (int i = 0; i < n; i++) {
        sideSum += matrix[i][n - 1 - i];
    }
    
    // Выводим результаты
    cout << "\nСумма главной диагонали: " << mainSum << endl;
    cout << "Сумма побочной диагонали: " << sideSum << endl;
    
    return 0;
}
```
