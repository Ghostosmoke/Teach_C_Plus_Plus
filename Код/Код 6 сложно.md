### main.cpp
```cpp
#include <iostream>
#include <limits>
#include <string>

using namespace std;

// Функция для ввода числа с проверкой
int inputNumber(const string& prompt) {
    int number;
    while (true) {
        cout << prompt;
        cin >> number;
        
        // Проверяем, что ввод корректный
        if (cin.fail()) {
            cin.clear();
            cin.ignore(numeric_limits<streamsize>::max(), '\n');
            cout << "Ошибка: введите число!" << endl;
            continue;
        }
        
        // Очищаем остаток строки
        cin.ignore(numeric_limits<streamsize>::max(), '\n');
        break;
    }
    return number;
}

// Функция для показа массива
void showArray(const int* arr, int size) {
    if (size == 0) {
        cout << "Массив пуст!" << endl;
    }
    else {
        cout << "Массив: ";
        for (int i = 0; i < size; i++) {
            cout << arr[i] << " ";
        }
        cout << endl;
    }
}

// Функция для добавления элемента в конец
int* addElement(int* oldArray, int oldSize, int newElement, int& newSize) {
    newSize = oldSize + 1;
    int* newArray = new int[newSize];
    
    // Копируем старые элементы
    for (int i = 0; i < oldSize; i++) {
        newArray[i] = oldArray[i];
    }
    
    // Добавляем новый элемент в конец
    newArray[oldSize] = newElement;
    
    // Удаляем старый массив (если он был)
    if (oldArray != nullptr) {
        delete[] oldArray;
    }
    
    return newArray;
}

// Функция для удаления элемента по номеру
int* removeByPosition(int* oldArray, int oldSize, int position, int& newSize) {
    // Проверяем корректность позиции
    if (position < 1 || position > oldSize) {
        cout << "Ошибка: неверный номер!" << endl;
        newSize = oldSize;
        return oldArray;
    }
    
    int index = position - 1;
    newSize = oldSize - 1;
    
    // Если массив станет пустым
    if (newSize == 0) {
        delete[] oldArray;
        return nullptr;
    }
    
    int* newArray = new int[newSize];
    
    // Копируем элементы до удаляемого
    for (int i = 0; i < index; i++) {
        newArray[i] = oldArray[i];
    }
    
    // Копируем элементы после удаляемого (со сдвигом)
    for (int i = index + 1; i < oldSize; i++) {
        newArray[i - 1] = oldArray[i];
    }
    
    delete[] oldArray;
    return newArray;
}

// Функция для удаления элемента по значению
int* removeByValue(int* oldArray, int oldSize, int value, int& newSize) {
    // Ищем элемент с таким значением
    int index = -1;
    for (int i = 0; i < oldSize; i++) {
        if (oldArray[i] == value) {
            index = i;
            break;
        }
    }
    
    // Если не нашли
    if (index == -1) {
        cout << "Элемент со значением " << value << " не найден!" << endl;
        newSize = oldSize;
        return oldArray;
    }
    
    newSize = oldSize - 1;
    
    // Если массив станет пустым
    if (newSize == 0) {
        delete[] oldArray;
        return nullptr;
    }
    
    int* newArray = new int[newSize];
    
    // Копируем элементы до найденного
    for (int i = 0; i < index; i++) {
        newArray[i] = oldArray[i];
    }
    
    // Копируем элементы после найденного (со сдвигом)
    for (int i = index + 1; i < oldSize; i++) {
        newArray[i - 1] = oldArray[i];
    }
    
    delete[] oldArray;
    cout << "Элемент " << value << " удалён!" << endl;
    return newArray;
}

// Функция для вставки элемента на позицию
int* insertElement(int* oldArray, int oldSize, int newElement, int position, int& newSize) {
    // Проверяем корректность позиции
    if (position < 1 || position > oldSize + 1) {
        cout << "Ошибка: неверная позиция!" << endl;
        newSize = oldSize;
        return oldArray;
    }
    
    int index = position - 1;
    newSize = oldSize + 1;
    int* newArray = new int[newSize];
    
    // Копируем элементы до позиции вставки
    for (int i = 0; i < index; i++) {
        newArray[i] = oldArray[i];
    }
    
    // Вставляем новый элемент
    newArray[index] = newElement;
    
    // Копируем остальные элементы со сдвигом
    for (int i = index; i < oldSize; i++) {
        newArray[i + 1] = oldArray[i];
    }
    
    // Удаляем старый массив (если он был)
    if (oldArray != nullptr) {
        delete[] oldArray;
    }
    
    cout << "Элемент " << newElement << " вставлен на позицию " << position << "!" << endl;
    return newArray;
}

// Функция для очистки массива
int* clearArray(int* oldArray, int& newSize) {
    if (oldArray != nullptr) {
        delete[] oldArray;
    }
    newSize = 0;
    cout << "Массив очищен!" << endl;
    return nullptr;
}

// Функция для показа меню
void showMenu() {
    cout << endl;
    cout << "=== Динамический массив ===" << endl;
    cout << "1. Показать массив" << endl;
    cout << "2. Добавить элемент в конец" << endl;
    cout << "3. Удалить элемент по номеру" << endl;
    cout << "4. Удалить элемент по значению" << endl;
    cout << "5. Вставить элемент на позицию" << endl;
    cout << "6. Очистить массив" << endl;
    cout << "0. Выход" << endl;
    cout << endl;
}

int main() {
    setlocale(LC_ALL, "Russian");
    
    int* arr = nullptr;
    int size = 0;
    
    cout << "=== Добро пожаловать в программу работы с динамическим массивом ===" << endl;
    
    // Главный цикл программы
    while (true) {
        // Показываем текущий размер
        cout << "Текущий размер массива: " << size << endl;
        
        // Показываем меню
        showMenu();
        
        // Запрашиваем выбор пользователя
        int choice = inputNumber("Выберите действие: ");
        
        // Обработка выбора
        if (choice == 0) {
            // Выход из программы
            cout << "До свидания!" << endl;
            break;
        }
        else if (choice == 1) {
            // Показать массив
            showArray(arr, size);
        }
        else if (choice == 2) {
            // Добавить элемент в конец
            if (size == 0) {
                cout << "Массив пуст, будет создан новый массив." << endl;
            }
            int newElement = inputNumber("Введите число для добавления: ");
            int newSize;
            arr = addElement(arr, size, newElement, newSize);
            size = newSize;
            cout << "Элемент " << newElement << " добавлен!" << endl;
        }
        else if (choice == 3) {
            // Удалить элемент по номеру
            if (size == 0) {
                cout << "Массив пуст, нечего удалять!" << endl;
            }
            else {
                int position = inputNumber("Какой элемент удалить (от 1 до " + to_string(size) + ")? ");
                int newSize;
                arr = removeByPosition(arr, size, position, newSize);
                size = newSize;
                if (size >= 0) {
                    cout << "Элемент удалён!" << endl;
                }
            }
        }
        else if (choice == 4) {
            // Удалить элемент по значению
            if (size == 0) {
                cout << "Массив пуст, нечего удалять!" << endl;
            }
            else {
                int value = inputNumber("Какое значение удалить? ");
                int newSize;
                arr = removeByValue(arr, size, value, newSize);
                size = newSize;
            }
        }
        else if (choice == 5) {
            // Вставить элемент на позицию
            int newElement = inputNumber("Какое число вставить? ");
            int position = inputNumber("На какую позицию (от 1 до " + to_string(size + 1) + ")? ");
            int newSize;
            arr = insertElement(arr, size, newElement, position, newSize);
            size = newSize;
        }
        else if (choice == 6) {
            // Очистить массив
            if (size == 0) {
                cout << "Массив уже пуст!" << endl;
            }
            else {
                arr = clearArray(arr, size);
            }
        }
        else {
            // Неверный выбор
            cout << "Ошибка: неверный выбор!" << endl;
        }
    }
    
    // Перед завершением программы освобождаем память (если массив не пуст)
    if (arr != nullptr) {
        delete[] arr;
    }
    
    return 0;
}
```
