## Вариант А: Простой (для новичков)

```cpp
#include <iostream>
using namespace std;

int main() {
    // Переменные для размеров матрицы
    int n, m;
    
    // Спрашиваем у пользователя размеры
    cout << "Введите количество строк: ";
    cin >> n;
    cout << "Введите количество столбцов: ";
    cin >> m;
    
    // Проверяем, что размеры правильные
    if (n < 1 || n > 10 || m < 1 || m > 10) {
        cout << "Ошибка: размеры должны быть от 1 до 10!" << endl;
        return 1;
    }
    
    // Создаём исходную матрицу
    int matrix[10][10];
    
    // Создаём матрицу для результата (транспонированную)
    int transposed[10][10];
    
    // Просим пользователя ввести элементы матрицы
    cout << "Введите элементы матрицы (" << n << "x" << m << "):" << endl;
    for (int i = 0; i < n; i++) {
        for (int j = 0; j < m; j++) {
            cin >> matrix[i][j];
        }
    }
    
    // Выводим исходную матрицу
    cout << "\nИсходная матрица (" << n << "x" << m << "):" << endl;
    for (int i = 0; i < n; i++) {
        for (int j = 0; j < m; j++) {
            cout << matrix[i][j] << "\t";
        }
        cout << endl;
    }
    
    // Транспонируем матрицу
    // Элемент [i][j] становится [j][i]
    for (int i = 0; i < n; i++) {
        for (int j = 0; j < m; j++) {
            transposed[j][i] = matrix[i][j];
        }
    }
    
    // Выводим транспонированную матрицу
    // Теперь у неё m строк и n столбцов
    cout << "\nТранспонированная матрица (" << m << "x" << n << "):" << endl;
    for (int i = 0; i < m; i++) {
        for (int j = 0; j < n; j++) {
            cout << transposed[i][j] << "\t";
        }
        cout << endl;
    }
    
    return 0;
}
```
