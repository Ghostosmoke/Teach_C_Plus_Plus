### main.cpp
```cpp
#include <iostream>
#include <limits>
#include <string>

using namespace std;

// Функция для ввода числа с проверкой
int inputNumber(const string& prompt) {
    int number;
    while (true) {
        cout << prompt;
        cin >> number;
        
        if (cin.fail()) {
            cin.clear();
            cin.ignore(numeric_limits<streamsize>::max(), '\n');
            cout << "Ошибка: введите число!" << endl;
            continue;
        }
        
        cin.ignore(numeric_limits<streamsize>::max(), '\n');
        break;
    }
    return number;
}

// Функция для создания матрицы
int** createMatrix(int rows, int cols) {
    int** matrix = new int*[rows];
    for (int i = 0; i < rows; i++) {
        matrix[i] = new int[cols];
    }
    return matrix;
}

// Функция для заполнения матрицы
void fillMatrix(int** matrix, int rows, int cols) {
    cout << "Введите " << rows * cols << " чисел для матрицы:" << endl;
    for (int i = 0; i < rows; i++) {
        for (int j = 0; j < cols; j++) {
            matrix[i][j] = inputNumber("Элемент [" + to_string(i) + "][" + to_string(j) + "]: ");
        }
    }
}

// Функция для вывода матрицы
void printMatrix(int** matrix, int rows, int cols, const string& label = "") {
    if (!label.empty()) {
        cout << label << ":" << endl;
    }
    for (int i = 0; i < rows; i++) {
        for (int j = 0; j < cols; j++) {
            cout << matrix[i][j] << "\t";
        }
        cout << endl;
    }
}

// Функция для освобождения памяти матрицы
void deleteMatrix(int** matrix, int rows) {
    for (int i = 0; i < rows; i++) {
        delete[] matrix[i];
    }
    delete[] matrix;
}

// Функция для вычисления суммы всех элементов
int calculateSum(int** matrix, int rows, int cols) {
    int sum = 0;
    for (int i = 0; i < rows; i++) {
        for (int j = 0; j < cols; j++) {
            sum += matrix[i][j];
        }
    }
    return sum;
}

// Функция для поиска максимального элемента
int findMax(int** matrix, int rows, int cols) {
    int max = matrix[0][0];
    for (int i = 0; i < rows; i++) {
        for (int j = 0; j < cols; j++) {
            if (matrix[i][j] > max) {
                max = matrix[i][j];
            }
        }
    }
    return max;
}

// Функция для поиска минимального элемента
int findMin(int** matrix, int rows, int cols) {
    int min = matrix[0][0];
    for (int i = 0; i < rows; i++) {
        for (int j = 0; j < cols; j++) {
            if (matrix[i][j] < min) {
                min = matrix[i][j];
            }
        }
    }
    return min;
}

// Функция для подсчёта чётных и нечётных элементов
void countEvenOdd(int** matrix, int rows, int cols, int& evenCount, int& oddCount) {
    evenCount = 0;
    oddCount = 0;
    
    for (int i = 0; i < rows; i++) {
        for (int j = 0; j < cols; j++) {
            if (matrix[i][j] % 2 == 0) {
                evenCount++;
            }
            else {
                oddCount++;
            }
        }
    }
}

// Функция для вывода результатов анализа
void printAnalysis(int sum, int max, int min, int evenCount, int oddCount) {
    cout << endl << "=== Результаты анализа ===" << endl;
    cout << "Сумма всех элементов: " << sum << endl;
    cout << "Максимальный элемент: " << max << endl;
    cout << "Минимальный элемент: " << min << endl;
    cout << "Количество чётных элементов: " << evenCount << endl;
    cout << "Количество нечётных элементов: " << oddCount << endl;
}

int main() {
    setlocale(LC_ALL, "Russian");
    
    cout << "=== Анализ матрицы ===" << endl;
    
    // Запрос размеров с проверкой
    int rows, cols;
    do {
        rows = inputNumber("Введите количество строк (1-10): ");
        if (rows < 1 || rows > 10) {
            cout << "Ошибка: количество строк должно быть от 1 до 10!" << endl;
        }
    } while (rows < 1 || rows > 10);
    
    do {
        cols = inputNumber("Введите количество столбцов (1-10): ");
        if (cols < 1 || cols > 10) {
            cout << "Ошибка: количество столбцов должно быть от 1 до 10!" << endl;
        }
    } while (cols < 1 || cols > 10);
    
    // Создание матрицы
    int** matrix = createMatrix(rows, cols);
    
    // Заполнение матрицы
    fillMatrix(matrix, rows, cols);
    
    // Вывод матрицы
    cout << endl;
    printMatrix(matrix, rows, cols, "Ваша матрица");
    
    // Анализ матрицы
    int sum = calculateSum(matrix, rows, cols);
    int max = findMax(matrix, rows, cols);
    int min = findMin(matrix, rows, cols);
    
    int evenCount, oddCount;
    countEvenOdd(matrix, rows, cols, evenCount, oddCount);
    
    // Вывод результатов
    printAnalysis(sum, max, min, evenCount, oddCount);
    
    // Освобождение памяти
    deleteMatrix(matrix, rows);
    
    cout << endl << "До свидания!" << endl;
    
    return 0;
}
```
