## Вариант А: Простой (для новичков)

```cpp
#include <iostream>
using namespace std;

int main() {
    // Переменная для размера матрицы
    int n;
    
    // Спрашиваем у пользователя размер матрицы
    cout << "Введите размер матрицы: ";
    cin >> n;
    
    // Проверяем, что размер правильный
    if (n < 1 || n > 10) {
        cout << "Ошибка: размер должен быть от 1 до 10!" << endl;
        return 1;
    }
    
    // Создаём статический массив
    int matrix[10][10];
    
    // Создаём матрицу для результата
    int result[10][10];
    
    // Просим пользователя ввести элементы матрицы
    cout << "Введите элементы матрицы (" << n << "x" << n << "):" << endl;
    for (int i = 0; i < n; i++) {
        for (int j = 0; j < n; j++) {
            cin >> matrix[i][j];
        }
    }
    
    // Выводим исходную матрицу
    cout << "\nИсходная матрица:" << endl;
    for (int i = 0; i < n; i++) {
        for (int j = 0; j < n; j++) {
            cout << matrix[i][j] << "\t";
        }
        cout << endl;
    }
    
    // Поворачиваем матрицу на 90 градусов по часовой стрелке
    // Формула: result[j][n-1-i] = matrix[i][j]
    for (int i = 0; i < n; i++) {
        for (int j = 0; j < n; j++) {
            result[j][n - 1 - i] = matrix[i][j];
        }
    }
    
    // Выводим повёрнутую матрицу
    cout << "\nПосле поворота на 90 градусов по часовой стрелке:" << endl;
    for (int i = 0; i < n; i++) {
        for (int j = 0; j < n; j++) {
            cout << result[i][j] << "\t";
        }
        cout << endl;
    }
    
    return 0;
}
```
