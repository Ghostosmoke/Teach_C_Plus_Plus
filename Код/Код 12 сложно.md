```cpp
#include <iostream>
#include <limits>
using namespace std;

// Функция для ввода размера с проверкой
int inputSize(const string& prompt, int max) {
    int size;
    while (true) {
        cout << prompt;
        cin >> size;
        
        if (cin.fail()) {
            cin.clear();
            cin.ignore(numeric_limits<streamsize>::max(), '\n');
            cout << "Ошибка: введите число!" << endl;
            continue;
        }
        
        if (size >= 1 && size <= max) {
            break;
        }
        
        cout << "Ошибка: размер должен быть от 1 до " << max << "!" << endl;
    }
    return size;
}

// Функция для заполнения матрицы
void fillMatrix(int matrix[][10], int n, int m) {
    for (int i = 0; i < n; i++) {
        cout << "Строка " << (i + 1) << ": ";
        for (int j = 0; j < m; j++) {
            cin >> matrix[i][j];
        }
    }
}

// Функция для вывода матрицы
void printMatrix(int matrix[][10], int n, int m) {
    for (int i = 0; i < n; i++) {
        for (int j = 0; j < m; j++) {
            cout << matrix[i][j] << "\t";
        }
        cout << endl;
    }
}

// Функция для зеркального отражения матрицы
void mirrorMatrix(int matrix[][10], int n, int m) {
    for (int i = 0; i < n; i++) {
        for (int j = 0; j < m / 2; j++) {
            // Меняем местами элементы
            int temp = matrix[i][j];
            matrix[i][j] = matrix[i][m - 1 - j];
            matrix[i][m - 1 - j] = temp;
        }
    }
}

// Функция для обработки одной матрицы
void processMatrix() {
    int n = inputSize("Введите количество строк: ", 10);
    int m = inputSize("Введите количество столбцов: ", 10);
    
    int matrix[10][10];
    
    cout << "Введите элементы матрицы (" << n << "x" << m << "):" << endl;
    fillMatrix(matrix, n, m);
    
    cout << "\nИсходная матрица:" << endl;
    printMatrix(matrix, n, m);
    
    mirrorMatrix(matrix, n, m);
    
    cout << "\nПосле зеркального отражения:" << endl;
    printMatrix(matrix, n, m);
}

int main() {
    setlocale(LC_ALL, "Russian");
    
    cout << "=== Зеркальное отражение матрицы ===" << endl;
    
    while (true) {
        processMatrix();
        
        cout << "\nХотите обработать ещё одну матрицу? (1 - да, 0 - нет): ";
        int choice;
        cin >> choice;
        
        if (choice == 0) {
            cout << "До свидания!" << endl;
            break;
        }
    }
    
    return 0;
}
```
