# Задача 3: Добавление элемента в массив

## Вариант А: Простой (для новичков)

### main.cpp
```cpp
#include <iostream>

using namespace std;

int main() {
    // Размер исходного массива
    int size = 5;
    
    // Создаём массив из 5 элементов
    int* arr = new int[size];
    
    // Просим пользователя заполнить массив
    cout << "Введите " << size << " чисел:" << endl;
    for (int i = 0; i < size; i++) {
        cout << "Число " << (i + 1) << ": ";
        cin >> arr[i];
    }
    
    // Выводим исходный массив
    cout << endl << "Исходный массив: ";
    for (int i = 0; i < size; i++) {
        cout << arr[i] << " ";
    }
    cout << endl;
    
    // Спрашиваем, какое число добавить
    int newElement;
    cout << "Какое число добавить? ";
    cin >> newElement;
    
    // Создаём новый массив на 1 элемент больше
    int newSize = size + 1;
    int* newArr = new int[newSize];
    
    // Копируем все элементы из старого массива в новый
    for (int i = 0; i < size; i++) {
        newArr[i] = arr[i];
    }
    
    // Добавляем новое число в конец
    newArr[size] = newElement;
    
    // Освобождаем старый массив
    delete[] arr;
    
    // Выводим новый массив
    cout << endl << "Новый массив: ";
    for (int i = 0; i < newSize; i++) {
        cout << newArr[i] << " ";
    }
    cout << endl;
    
    // Освобождаем новый массив
    delete[] newArr;
    
    return 0;
}
```
