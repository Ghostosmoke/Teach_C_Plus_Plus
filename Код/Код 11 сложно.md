```cpp
#include <iostream>
#include <limits>
using namespace std;

// Функция для ввода размера с проверкой
int inputSize(const string& prompt, int max) {
    int size;
    while (true) {
        cout << prompt;
        cin >> size;
        
        if (cin.fail()) {
            cin.clear();
            cin.ignore(numeric_limits<streamsize>::max(), '\n');
            cout << "Ошибка: введите число!" << endl;
            continue;
        }
        
        if (size >= 1 && size <= max) {
            break;
        }
        
        cout << "Ошибка: размер должен быть от 1 до " << max << "!" << endl;
    }
    return size;
}

// Функция для заполнения матрицы
void fillMatrix(int matrix[][10], int n, int m) {
    for (int i = 0; i < n; i++) {
        cout << "Строка " << (i + 1) << ": ";
        for (int j = 0; j < m; j++) {
            cin >> matrix[i][j];
        }
    }
}

// Функция для вывода матрицы
void printMatrix(int matrix[][10], int n, int m) {
    for (int i = 0; i < n; i++) {
        for (int j = 0; j < m; j++) {
            cout << matrix[i][j] << "\t";
        }
        cout << endl;
    }
}

// Функция для сложения двух матриц
void addMatrices(int a[][10], int b[][10], int c[][10], int n, int m) {
    for (int i = 0; i < n; i++) {
        for (int j = 0; j < m; j++) {
            c[i][j] = a[i][j] + b[i][j];
        }
    }
}

// Функция для обработки одной пары матриц
void processMatrices() {
    int n = inputSize("Введите количество строк: ", 10);
    int m = inputSize("Введите количество столбцов: ", 10);
    
    int a[10][10];
    int b[10][10];
    int c[10][10];
    
    cout << "\nВведите матрицу A (" << n << "x" << m << "):" << endl;
    fillMatrix(a, n, m);
    
    cout << "\nВведите матрицу B (" << n << "x" << m << "):" << endl;
    fillMatrix(b, n, m);
    
    addMatrices(a, b, c, n, m);
    
    cout << "\nМатрица A:" << endl;
    printMatrix(a, n, m);
    
    cout << "\nМатрица B:" << endl;
    printMatrix(b, n, m);
    
    cout << "\nРезультат (A + B):" << endl;
    printMatrix(c, n, m);
}

int main() {
    setlocale(LC_ALL, "Russian");
    
    cout << "=== Сложение двух матриц ===" << endl;
    
    while (true) {
        processMatrices();
        
        cout << "\nХотите обработать ещё одну пару матриц? (1 - да, 0 - нет): ";
        int choice;
        cin >> choice;
        
        if (choice == 0) {
            cout << "До свидания!" << endl;
            break;
        }
    }
    
    return 0;
}
```
