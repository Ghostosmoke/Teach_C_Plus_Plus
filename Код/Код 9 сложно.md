```cpp
#include <iostream>
#include <limits>
using namespace std;

// Функция для ввода размера с проверкой
int inputSize(const string& prompt, int max) {
    int size;
    while (true) {
        cout << prompt;
        cin >> size;
        
        // Проверяем, что ввод корректный
        if (cin.fail()) {
            cin.clear();
            cin.ignore(numeric_limits<streamsize>::max(), '\n');
            cout << "Ошибка: введите число!" << endl;
            continue;
        }
        
        // Проверяем диапазон
        if (size >= 1 && size <= max) {
            break;
        }
        
        cout << "Ошибка: размер должен быть от 1 до " << max << "!" << endl;
    }
    return size;
}

// Функция для заполнения матрицы
void fillMatrix(int matrix[][10], int n, int m) {
    cout << "Введите элементы матрицы (" << n << "x" << m << "):" << endl;
    for (int i = 0; i < n; i++) {
        cout << "Строка " << (i + 1) << ": ";
        for (int j = 0; j < m; j++) {
            cin >> matrix[i][j];
        }
    }
}

// Функция для вывода матрицы
void printMatrix(int matrix[][10], int n, int m) {
    for (int i = 0; i < n; i++) {
        for (int j = 0; j < m; j++) {
            cout << matrix[i][j] << "\t";
        }
        cout << endl;
    }
}

// Функция для транспонирования матрицы
void transposeMatrix(int source[][10], int result[][10], int n, int m) {
    for (int i = 0; i < n; i++) {
        for (int j = 0; j < m; j++) {
            result[j][i] = source[i][j];
        }
    }
}

// Функция для обработки одной матрицы
void processMatrix() {
    // Ввод размеров
    int n = inputSize("Введите количество строк: ", 10);
    int m = inputSize("Введите количество столбцов: ", 10);
    
    // Объявляем матрицы
    int matrix[10][10];
    int transposed[10][10];
    
    // Заполнение исходной матрицы
    fillMatrix(matrix, n, m);
    
    // Вывод исходной матрицы
    cout << "\nИсходная матрица (" << n << "x" << m << "):" << endl;
    printMatrix(matrix, n, m);
    
    // Транспонирование
    transposeMatrix(matrix, transposed, n, m);
    
    // Вывод результата
    cout << "\nТранспонированная матрица (" << m << "x" << n << "):" << endl;
    printMatrix(transposed, m, n);
}

int main() {
    setlocale(LC_ALL, "Russian");
    
    cout << "=== Транспонирование матрицы ===" << endl;
    
    // Главный цикл программы
    while (true) {
        processMatrix();
        
        // Спрашиваем, хочет ли пользователь продолжить
        cout << "\nХотите обработать ещё одну матрицу? (1 - да, 0 - нет): ";
        int choice;
        cin >> choice;
        
        if (choice == 0) {
            cout << "До свидания!" << endl;
            break;
        }
    }
    
    return 0;
}
```
