#include <iostream>
#include <limits>

using namespace std;

// Функция для ввода числа с проверкой
int inputNumber(const string& prompt) {
    int number;
    while (true) {
        cout << prompt;
        cin >> number;
        
        // Проверяем, что ввод корректный
        if (cin.fail()) {
            cin.clear();
            cin.ignore(numeric_limits<streamsize>::max(), '\n');
            cout << "Ошибка: введите число!" << endl;
            continue;
        }
        
        // Очищаем остаток строки
        cin.ignore(numeric_limits<streamsize>::max(), '\n');
        break;
    }
    return number;
}

// Функция для заполнения массива
void fillArray(int* arr, int size) {
    cout << "Введите " << size << " чисел:" << endl;
    for (int i = 0; i < size; i++) {
        arr[i] = inputNumber("Число " + to_string(i + 1) + ": ");
    }
}

// Функция для вывода массива
void printArray(const int* arr, int size, const string& label = "") {
    if (!label.empty()) {
        cout << label << ": ";
    }
    for (int i = 0; i < size; i++) {
        cout << arr[i] << " ";
    }
    cout << endl;
}

// Функция для добавления элемента в конец массива
// Возвращает указатель на новый массив и обновляет размер
int* addElement(int* oldArray, int oldSize, int newElement, int& newSize) {
    // Создаём новый массив на 1 элемент больше
    newSize = oldSize + 1;
    int* newArray = new int[newSize];
    
    // Копируем все элементы из старого массива
    for (int i = 0; i < oldSize; i++) {
        newArray[i] = oldArray[i];
    }
    
    // Добавляем новый элемент в конец
    newArray[oldSize] = newElement;
    
    // Освобождаем старый массив
    delete[] oldArray;
    
    // Возвращаем новый массив
    return newArray;
}

// Функция для добавления элемента на любую позицию
int* insertElement(int* oldArray, int oldSize, int newElement, int position, int& newSize) {
    // Проверяем корректность позиции
    if (position < 0 || position > oldSize) {
        cout << "Ошибка: неверная позиция!" << endl;
        newSize = oldSize;
        return oldArray;
    }
    
    // Создаём новый массив на 1 элемент больше
    newSize = oldSize + 1;
    int* newArray = new int[newSize];
    
    // Копируем элементы до позиции вставки
    for (int i = 0; i < position; i++) {
        newArray[i] = oldArray[i];
    }
    
    // Вставляем новый элемент
    newArray[position] = newElement;
    
    // Копируем остальные элементы со сдвигом
    for (int i = position; i < oldSize; i++) {
        newArray[i + 1] = oldArray[i];
    }
    
    // Освобождаем старый массив
    delete[] oldArray;
    
    // Возвращаем новый массив
    return newArray;
}

// Функция для обработки одного массива
void processArray() {
    // Запрашиваем размер массива
    int size = inputNumber("Введите размер массива (от 1 до 100): ");
    
    // Проверяем размер
    while (size < 1 || size > 100) {
        cout << "Ошибка: размер должен быть от 1 до 100!" << endl;
        size = inputNumber("Введите размер массива: ");
    }
    
    // Создаём массив
    int* arr = new int[size];
    
    // Заполняем массив
    fillArray(arr, size);
    
    // Выводим исходный массив
    printArray(arr, size, "Исходный массив");
    
    // Цикл для добавления элементов
    while (true) {
        cout << endl;
        cout << "1. Добавить элемент в конец" << endl;
        cout << "2. Вставить элемент на позицию" << endl;
        cout << "3. Завершить" << endl;
        cout << "Выберите действие: ";
        
        int choice = inputNumber("");
        
        if (choice == 3) {
            break;
        }
        
        if (choice == 1) {
            // Добавление в конец
            int newElement = inputNumber("Какое число добавить? ");
            int newSize;
            arr = addElement(arr, size, newElement, newSize);
            size = newSize;
            printArray(arr, size, "Новый массив");
        }
        else if (choice == 2) {
            // Вставка на позицию
            int newElement = inputNumber("Какое число вставить? ");
            int position = inputNumber("На какую позицию (от 1 до " + to_string(size + 1) + ")? ") - 1;
            int newSize;
            arr = insertElement(arr, size, newElement, position, newSize);
            size = newSize;
            printArray(arr, size, "Новый массив");
        }
        else {
            cout << "Ошибка: неверный выбор!" << endl;
        }
    }
    
    // Освобождаем память
    delete[] arr;
}

int main() {
    setlocale(LC_ALL, "Russian");
    
    cout << "=== Добавление элементов в массив ===" << endl;
    
    // Главный цикл программы
    while (true) {
        processArray();
        
        cout << endl;
        cout << "Хотите обработать ещё один массив? (1 - да, 0 - нет): ";
        int choice = inputNumber("");
        
        if (choice == 0) {
            cout << "До свидания!" << endl;
            break;
        }
    }
    
    return 0;
}
