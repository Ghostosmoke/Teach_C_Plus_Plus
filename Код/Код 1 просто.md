### main.cpp
```cpp
#include <iostream>

using namespace std;

int main() {
    // Объявляем переменные для чисел
    double num1, num2;
    char operation;
    
    // Просим пользователя ввести первое число
    cout << "Введите первое число: ";
    cin >> num1;
    
    // Просим ввести второе число
    cout << "Введите второе число: ";
    cin >> num2;
    
    // Просим выбрать операцию
    cout << "Выберите операцию (+, -, *, /): ";
    cin >> operation;
    
    // Проверяем, какая операция выбрана
    if (operation == '+') {
        // Складываем и выводим результат
        cout << "Результат: " << num1 + num2 << endl;
    }
    else if (operation == '-') {
        // Вычитаем и выводим результат
        cout << "Результат: " << num1 - num2 << endl;
    }
    else if (operation == '*') {
        // Умножаем и выводим результат
        cout << "Результат: " << num1 * num2 << endl;
    }
    else if (operation == '/') {
        // Проверяем, что не делим на ноль
        if (num2 == 0) {
            cout << "Ошибка: деление на ноль!" << endl;
        }
        else {
            // Делим и выводим результат
            cout << "Результат: " << num1 / num2 << endl;
        }
    }
    else {
        // Если ввели что-то другое
        cout << "Ошибка: неизвестная операция!" << endl;
    }
    
    return 0;
}
```
