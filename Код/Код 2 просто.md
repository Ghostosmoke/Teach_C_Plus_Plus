### main.cpp
```cpp
#include <iostream>

using namespace std;

int main() {
    // Переменная для размера массива
    int size;
    
    // Спрашиваем у пользователя размер массива
    cout << "Сколько чисел будет в массиве? ";
    cin >> size;
    
    // Проверяем, что размер правильный
    if (size < 1 || size > 100) {
        cout << "Ошибка: размер должен быть от 1 до 100!" << endl;
        return 1;  // завершаем программу с ошибкой
    }
    
    // Создаём динамический массив
    // new выделяет память в куче и возвращает указатель на неё
    int* arr = new int[size];
    
    // Просим пользователя ввести числа
    cout << "Введите " << size << " чисел:" << endl;
    for (int i = 0; i < size; i++) {
        cout << "Число " << (i + 1) << ": ";
        cin >> arr[i];  // arr[i] - это i-й элемент массива
    }
    
    // Вычисляем сумму всех чисел
    int sum = 0;
    for (int i = 0; i < size; i++) {
        sum = sum + arr[i];  // добавляем каждое число к сумме
    }
    
    // Вычисляем среднее арифметическое
    double average = (double)sum / size;  // делим сумму на количество чисел
    // (double)sum - преобразуем int в double, чтобы деление было точным
    
    // Ищем максимум и минимум
    int max = arr[0];  // предполагаем, что первый элемент - максимум
    int min = arr[0];  // предполагаем, что первый элемент - минимум
    
    for (int i = 1; i < size; i++) {  // начинаем с 1, так как 0-й уже проверили
        // Проверяем, больше ли текущий элемент текущего максимума
        if (arr[i] > max) {
            max = arr[i];  // обновляем максимум
        }
        
        // Проверяем, меньше ли текущий элемент текущего минимума
        if (arr[i] < min) {
            min = arr[i];  // обновляем минимум
        }
    }
    
    // Выводим результаты
    cout << endl;
    cout << "Сумма: " << sum << endl;
    cout << "Среднее: " << average << endl;
    cout << "Максимум: " << max << endl;
    cout << "Минимум: " << min << endl;
    
    // Освобождаем память!
    // Если этого не сделать, будет утечка памяти
    delete[] arr;
    
    return 0;
}
```
