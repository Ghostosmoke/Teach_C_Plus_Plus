## Вариант А: Простой (для новичков)

```cpp
#include <iostream>
using namespace std;

int main() {
    // Переменные для размеров матрицы
    int n, m;
    
    // Спрашиваем у пользователя размеры
    cout << "Введите количество строк: ";
    cin >> n;
    cout << "Введите количество столбцов: ";
    cin >> m;
    
    // Проверяем, что размеры правильные
    if (n < 1 || n > 10 || m < 1 || m > 10) {
        cout << "Ошибка: размеры должны быть от 1 до 10!" << endl;
        return 1;
    }
    
    // Создаём три матрицы: A, B и результат C
    int a[10][10];
    int b[10][10];
    int c[10][10];
    
    // Вводим матрицу A
    cout << "\nВведите матрицу A (" << n << "x" << m << "):" << endl;
    for (int i = 0; i < n; i++) {
        for (int j = 0; j < m; j++) {
            cin >> a[i][j];
        }
    }
    
    // Вводим матрицу B
    cout << "\nВведите матрицу B (" << n << "x" << m << "):" << endl;
    for (int i = 0; i < n; i++) {
        for (int j = 0; j < m; j++) {
            cin >> b[i][j];
        }
    }
    
    // Складываем матрицы поэлементно
    for (int i = 0; i < n; i++) {
        for (int j = 0; j < m; j++) {
            c[i][j] = a[i][j] + b[i][j];
        }
    }
    
    // Выводим матрицу A
    cout << "\nМатрица A:" << endl;
    for (int i = 0; i < n; i++) {
        for (int j = 0; j < m; j++) {
            cout << a[i][j] << "\t";
        }
        cout << endl;
    }
    
    // Выводим матрицу B
    cout << "\nМатрица B:" << endl;
    for (int i = 0; i < n; i++) {
        for (int j = 0; j < m; j++) {
            cout << b[i][j] << "\t";
        }
        cout << endl;
    }
    
    // Выводим результат
    cout << "\nРезультат (A + B):" << endl;
    for (int i = 0; i < n; i++) {
        for (int j = 0; j < m; j++) {
            cout << c[i][j] << "\t";
        }
        cout << endl;
    }
    
    return 0;
}
```
