
## Вариант А: Простой (для новичков)

### main.cpp
```cpp
#include <iostream>

using namespace std;

int main() {
    // Размер исходного массива
    int size = 7;
    
    // Создаём массив из 7 элементов
    int* arr = new int[size];
    
    // Заполняем массив числами
    // (для простоты заполняем сразу, без ввода от пользователя)
    arr[0] = 10;
    arr[1] = 20;
    arr[2] = 30;
    arr[3] = 40;
    arr[4] = 50;
    arr[5] = 60;
    arr[6] = 70;
    
    // Выводим исходный массив
    cout << "Исходный массив: ";
    for (int i = 0; i < size; i++) {
        cout << arr[i] << " ";
    }
    cout << endl;
    
    // Спрашиваем, какой элемент удалить (по номеру, начиная с 1)
    int position;
    cout << "Какой элемент удалить (по номеру от 1 до " << size << ")? ";
    cin >> position;
    
    // Проверяем, что номер правильный
    if (position < 1 || position > size) {
        cout << "Ошибка: неверный номер!" << endl;
        delete[] arr;  // освобождаем память перед выходом
        return 1;
    }
    
    // Преобразуем номер в индекс (индексы начинаются с 0)
    int index = position - 1;
    
    // Создаём новый массив на 1 элемент меньше
    int newSize = size - 1;
    int* newArr = new int[newSize];
    
    // Копируем элементы в новый массив, пропуская удаляемый
    // Сначала копируем всё, что до удаляемого элемента
    for (int i = 0; i < index; i++) {
        newArr[i] = arr[i];
    }
    
    // Потом копируем всё, что после удаляемого элемента
    // Обратите внимание: в новом массиве индекс на 1 меньше
    for (int i = index + 1; i < size; i++) {
        newArr[i - 1] = arr[i];
    }
    
    // Освобождаем старый массив
    delete[] arr;
    
    // Выводим новый массив
    cout << "Новый массив: ";
    for (int i = 0; i < newSize; i++) {
        cout << newArr[i] << " ";
    }
    cout << endl;
    
    // Освобождаем новый массив
    delete[] newArr;
    
    return 0;
}
```
