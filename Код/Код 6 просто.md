# Задача 6: Динамический массив с меню (итоговая задача)

Это итоговая задача, которая объединяет всё, что мы изучили в предыдущих задачах. Мы создадим программу с меню, которая умеет:
- Создавать массив
- Показывать его содержимое
- Добавлять элементы в конец
- Удалять элементы (по номеру и по значению)
- Вставлять элементы на любую позицию
- Очищать массив
- Завершать работу

---

## Вариант А: Простой (для новичков)

### main.cpp
```cpp
#include <iostream>

using namespace std;

int main() {
    // Указатель на массив (изначально пустой)
    int* arr = nullptr;  // nullptr - это "пустой указатель", означает "ничего"
    int size = 0;        // размер массива (изначально 0)
    
    // Главный цикл программы
    while (true) {
        // Показываем меню
        cout << endl;
        cout << "=== Динамический массив ===" << endl;
        cout << "Текущий размер: " << size << endl;
        cout << endl;
        cout << "1. Показать массив" << endl;
        cout << "2. Добавить элемент в конец" << endl;
        cout << "3. Удалить элемент по номеру" << endl;
        cout << "4. Удалить элемент по значению" << endl;
        cout << "5. Вставить элемент на позицию" << endl;
        cout << "6. Очистить массив" << endl;
        cout << "0. Выход" << endl;
        cout << endl;
        cout << "Выберите действие: ";
        
        int choice;
        cin >> choice;
        
        // Обработка выбора пользователя
        if (choice == 0) {
            // Выход из программы
            cout << "До свидания!" << endl;
            break;  // выходим из цикла while
        }
        else if (choice == 1) {
            // Показать массив
            if (size == 0) {
                cout << "Массив пуст!" << endl;
            }
            else {
                cout << "Массив: ";
                for (int i = 0; i < size; i++) {
                    cout << arr[i] << " ";
                }
                cout << endl;
            }
        }
        else if (choice == 2) {
            // Добавить элемент в конец
            int newElement;
            cout << "Введите число для добавления: ";
            cin >> newElement;
            
            // Создаём новый массив на 1 элемент больше
            int newSize = size + 1;
            int* newArr = new int[newSize];
            
            // Копируем старые элементы
            for (int i = 0; i < size; i++) {
                newArr[i] = arr[i];
            }
            
            // Добавляем новый элемент в конец
            newArr[size] = newElement;
            
            // Удаляем старый массив (если он был)
            if (arr != nullptr) {
                delete[] arr;
            }
            
            // Обновляем указатель и размер
            arr = newArr;
            size = newSize;
            
            cout << "Элемент " << newElement << " добавлен!" << endl;
        }
        else if (choice == 3) {
            // Удалить элемент по номеру
            if (size == 0) {
                cout << "Массив пуст, нечего удалять!" << endl;
            }
            else {
                int position;
                cout << "Какой элемент удалить (от 1 до " << size << ")? ";
                cin >> position;
                
                // Проверяем номер
                if (position < 1 || position > size) {
                    cout << "Ошибка: неверный номер!" << endl;
                }
                else {
                    int index = position - 1;  // преобразуем номер в индекс
                    
                    // Создаём новый массив на 1 элемент меньше
                    int newSize = size - 1;
                    int* newArr = new int[newSize];
                    
                    // Копируем элементы до удаляемого
                    for (int i = 0; i < index; i++) {
                        newArr[i] = arr[i];
                    }
                    
                    // Копируем элементы после удаляемого (со сдвигом)
                    for (int i = index + 1; i < size; i++) {
                        newArr[i - 1] = arr[i];
                    }
                    
                    // Удаляем старый массив
                    delete[] arr;
                    
                    // Обновляем указатель и размер
                    arr = newArr;
                    size = newSize;
                    
                    cout << "Элемент удалён!" << endl;
                }
            }
        }
        else if (choice == 4) {
            // Удалить элемент по значению
            if (size == 0) {
                cout << "Массив пуст, нечего удалять!" << endl;
            }
            else {
                int value;
                cout << "Какое значение удалить? ";
                cin >> value;
                
                // Ищем элемент с таким значением
                int index = -1;
                for (int i = 0; i < size; i++) {
                    if (arr[i] == value) {
                        index = i;
                        break;  // нашли, выходим из цикла
                    }
                }
                
                // Если не нашли
                if (index == -1) {
                    cout << "Элемент со значением " << value << " не найден!" << endl;
                }
                else {
                    // Создаём новый массив на 1 элемент меньше
                    int newSize = size - 1;
                    int* newArr = new int[newSize];
                    
                    // Копируем элементы до найденного
                    for (int i = 0; i < index; i++) {
                        newArr[i] = arr[i];
                    }
                    
                    // Копируем элементы после найденного (со сдвигом)
                    for (int i = index + 1; i < size; i++) {
                        newArr[i - 1] = arr[i];
                    }
                    
                    // Удаляем старый массив
                    delete[] arr;
                    
                    // Обновляем указатель и размер
                    arr = newArr;
                    size = newSize;
                    
                    cout << "Элемент " << value << " удалён!" << endl;
                }
            }
        }
        else if (choice == 5) {
            // Вставить элемент на позицию
            int newElement;
            cout << "Какое число вставить? ";
            cin >> newElement;
            
            int position;
            cout << "На какую позицию (от 1 до " << (size + 1) << ")? ";
            cin >> position;
            
            // Проверяем позицию
            if (position < 1 || position > size + 1) {
                cout << "Ошибка: неверная позиция!" << endl;
            }
            else {
                int index = position - 1;  // преобразуем позицию в индекс
                
                // Создаём новый массив на 1 элемент больше
                int newSize = size + 1;
                int* newArr = new int[newSize];
                
                // Копируем элементы до позиции вставки
                for (int i = 0; i < index; i++) {
                    newArr[i] = arr[i];
                }
                
                // Вставляем новый элемент
                newArr[index] = newElement;
                
                // Копируем остальные элементы со сдвигом
                for (int i = index; i < size; i++) {
                    newArr[i + 1] = arr[i];
                }
                
                // Удаляем старый массив (если он был)
                if (arr != nullptr) {
                    delete[] arr;
                }
                
                // Обновляем указатель и размер
                arr = newArr;
                size = newSize;
                
                cout << "Элемент " << newElement << " вставлен на позицию " << position << "!" << endl;
            }
        }
        else if (choice == 6) {
            // Очистить массив
            if (size == 0) {
                cout << "Массив уже пуст!" << endl;
            }
            else {
                // Удаляем массив
                delete[] arr;
                
                // Обнуляем указатель и размер
                arr = nullptr;
                size = 0;
                
                cout << "Массив очищен!" << endl;
            }
        }
        else {
            // Неверный выбор
            cout << "Ошибка: неверный выбор!" << endl;
        }
    }
    
    // Перед завершением программы освобождаем память (если массив не пуст)
    if (arr != nullptr) {
        delete[] arr;
    }
    
    return 0;
}
```
