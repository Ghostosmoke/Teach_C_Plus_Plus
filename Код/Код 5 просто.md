# Задача 5: Вставка элемента в любую позицию массива

---

## Вариант А: Простой (для новичков)

### main.cpp
```cpp
#include <iostream>

using namespace std;

int main() {
    // Размер исходного массива
    int size = 5;
    
    // Создаём массив из 5 элементов
    int* arr = new int[size];
    
    // Заполняем массив числами
    arr[0] = 10;
    arr[1] = 20;
    arr[2] = 30;
    arr[3] = 40;
    arr[4] = 50;
    
    // Выводим исходный массив
    cout << "Исходный массив: ";
    for (int i = 0; i < size; i++) {
        cout << arr[i] << " ";
    }
    cout << endl;
    
    // Спрашиваем, какое число вставить
    int newElement;
    cout << "Какое число вставить? ";
    cin >> newElement;
    
    // Спрашиваем, на какую позицию вставить (от 1 до size+1)
    int position;
    cout << "На какую позицию (от 1 до " << (size + 1) << ")? ";
    cin >> position;
    
    // Проверяем, что позиция правильная
    if (position < 1 || position > size + 1) {
        cout << "Ошибка: неверная позиция!" << endl;
        delete[] arr;  // освобождаем память перед выходом
        return 1;
    }
    
    // Преобразуем позицию в индекс (индексы начинаются с 0)
    int index = position - 1;
    
    // Создаём новый массив на 1 элемент больше
    int newSize = size + 1;
    int* newArr = new int[newSize];
    
    // Копируем элементы до позиции вставки
    for (int i = 0; i < index; i++) {
        newArr[i] = arr[i];
    }
    
    // Вставляем новый элемент
    newArr[index] = newElement;
    
    // Копируем остальные элементы со сдвигом вправо
    for (int i = index; i < size; i++) {
        newArr[i + 1] = arr[i];
    }
    
    // Освобождаем старый массив
    delete[] arr;
    
    // Выводим новый массив
    cout << "Новый массив: ";
    for (int i = 0; i < newSize; i++) {
        cout << newArr[i] << " ";
    }
    cout << endl;
    
    // Освобождаем новый массив
    delete[] newArr;
    
    return 0;
}
```
