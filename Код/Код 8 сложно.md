## Вариант Б: Улучшенный (для хорошего ученика)

```cpp
#include <iostream>
#include <limits>
using namespace std;

// Функция для ввода размера матрицы с проверкой
int inputSize() {
    int n;
    while (true) {
        cout << "Введите размер матрицы (1-10): ";
        cin >> n;
        
        // Проверяем, что ввод корректный
        if (cin.fail()) {
            // Очистка потока при ошибке ввода
            cin.clear();
            cin.ignore(numeric_limits<streamsize>::max(), '\n');
            cout << "Ошибка: введите число!" << endl;
            continue;
        }
        
        // Проверяем диапазон
        if (n >= 1 && n <= 10) {
            break;  // всё хорошо, выходим из цикла
        }
        
        cout << "Ошибка: размер должен быть от 1 до 10!" << endl;
    }
    return n;
}

// Функция для заполнения матрицы
void fillMatrix(int matrix[][10], int n) {
    cout << "Введите элементы матрицы (" << n << "x" << n << "):" << endl;
    for (int i = 0; i < n; i++) {
        cout << "Строка " << (i + 1) << ": ";
        for (int j = 0; j < n; j++) {
            cin >> matrix[i][j];
        }
    }
}

// Функция для вывода матрицы
void printMatrix(int matrix[][10], int n) {
    cout << "\nВаша матрица:" << endl;
    for (int i = 0; i < n; i++) {
        for (int j = 0; j < n; j++) {
            cout << matrix[i][j] << "\t";
        }
        cout << endl;
    }
}

// Функция для вычисления суммы главной диагонали
int calculateMainDiagonalSum(int matrix[][10], int n) {
    int sum = 0;
    for (int i = 0; i < n; i++) {
        sum += matrix[i][i];  // i == j для главной диагонали
    }
    return sum;
}

// Функция для вычисления суммы побочной диагонали
int calculateSideDiagonalSum(int matrix[][10], int n) {
    int sum = 0;
    for (int i = 0; i < n; i++) {
        sum += matrix[i][n - 1 - i];  // i + j == n - 1 для побочной диагонали
    }
    return sum;
}

// Функция для вывода результатов
void printResults(int mainSum, int sideSum) {
    cout << "\n=== Результаты ===" << endl;
    cout << "Сумма главной диагонали: " << mainSum << endl;
    cout << "Сумма побочной диагонали: " << sideSum << endl;
}

// Функция для обработки одной матрицы
void processMatrix() {
    // Ввод размера
    int n = inputSize();
    
    // Объявляем матрицу
    int matrix[10][10];
    
    // Заполнение матрицы
    fillMatrix(matrix, n);
    
    // Вывод матрицы
    printMatrix(matrix, n);
    
    // Вычисления
    int mainSum = calculateMainDiagonalSum(matrix, n);
    int sideSum = calculateSideDiagonalSum(matrix, n);
    
    // Вывод результатов
    printResults(mainSum, sideSum);
}

int main() {
    setlocale(LC_ALL, "Russian");  // для корректного отображения русского текста
    
    cout << "=== Анализ диагоналей матрицы ===" << endl;
    
    // Главный цикл программы
    while (true) {
        processMatrix();
        
        // Спрашиваем, хочет ли пользователь продолжить
        cout << "\nХотите обработать ещё одну матрицу? (1 - да, 0 - нет): ";
        int choice;
        cin >> choice;
        
        if (choice == 0) {
            cout << "До свидания!" << endl;
            break;
        }
    }
    
    return 0;
}
```
