## Вариант А: Простой (для новичков)

```cpp
#include <iostream>
using namespace std;

int main() {
    // Переменные для размеров матрицы
    int n, m;
    
    // Спрашиваем у пользователя размеры
    cout << "Введите количество строк: ";
    cin >> n;
    cout << "Введите количество столбцов: ";
    cin >> m;
    
    // Проверяем, что размеры правильные
    if (n < 1 || n > 10 || m < 1 || m > 10) {
        cout << "Ошибка: размеры должны быть от 1 до 10!" << endl;
        return 1;
    }
    
    // Создаём матрицу
    int matrix[10][10];
    
    // Просим пользователя ввести элементы матрицы
    cout << "Введите элементы матрицы (" << n << "x" << m << "):" << endl;
    for (int i = 0; i < n; i++) {
        for (int j = 0; j < m; j++) {
            cin >> matrix[i][j];
        }
    }
    
    // Выводим исходную матрицу
    cout << "\nИсходная матрица:" << endl;
    for (int i = 0; i < n; i++) {
        for (int j = 0; j < m; j++) {
            cout << matrix[i][j] << "\t";
        }
        cout << endl;
    }
    
    // Зеркально отражаем матрицу слева направо
    // Меняем местами элементы в каждой строке
    for (int i = 0; i < n; i++) {
        for (int j = 0; j < m / 2; j++) {
            // Меняем местами matrix[i][j] и matrix[i][m-1-j]
            int temp = matrix[i][j];
            matrix[i][j] = matrix[i][m - 1 - j];
            matrix[i][m - 1 - j] = temp;
        }
    }
    
    // Выводим отражённую матрицу
    cout << "\nПосле зеркального отражения:" << endl;
    for (int i = 0; i < n; i++) {
        for (int j = 0; j < m; j++) {
            cout << matrix[i][j] << "\t";
        }
        cout << endl;
    }
    
    return 0;
}
```
