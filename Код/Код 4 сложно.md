```cpp
#include <iostream>
#include <limits>
#include <string>

using namespace std;

// Функция для ввода числа с проверкой
int inputNumber(const string& prompt) {
    int number;
    while (true) {
        cout << prompt;
        cin >> number;
        
        // Проверяем, что ввод корректный
        if (cin.fail()) {
            cin.clear();
            cin.ignore(numeric_limits<streamsize>::max(), '\n');
            cout << "Ошибка: введите число!" << endl;
            continue;
        }
        
        // Очищаем остаток строки
        cin.ignore(numeric_limits<streamsize>::max(), '\n');
        break;
    }
    return number;
}

// Функция для заполнения массива
void fillArray(int* arr, int size) {
    cout << "Введите " << size << " чисел:" << endl;
    for (int i = 0; i < size; i++) {
        arr[i] = inputNumber("Число " + to_string(i + 1) + ": ");
    }
}

// Функция для вывода массива
void printArray(const int* arr, int size, const string& label = "") {
    if (!label.empty()) {
        cout << label << ": ";
    }
    for (int i = 0; i < size; i++) {
        cout << arr[i] << " ";
    }
    cout << endl;
}

// Функция для удаления элемента по номеру
// Возвращает указатель на новый массив и обновляет размер
int* removeElement(int* oldArray, int oldSize, int position, int& newSize) {
    // Проверяем корректность позиции (позиция от 1 до oldSize)
    if (position < 1 || position > oldSize) {
        cout << "Ошибка: неверный номер элемента!" << endl;
        newSize = oldSize;
        return oldArray;  // возвращаем старый массив без изменений
    }
    
    // Преобразуем номер в индекс
    int index = position - 1;
    
    // Создаём новый массив на 1 элемент меньше
    newSize = oldSize - 1;
    int* newArray = new int[newSize];
    
    // Копируем элементы до удаляемого
    for (int i = 0; i < index; i++) {
        newArray[i] = oldArray[i];
    }
    
    // Копируем элементы после удаляемого (со сдвигом)
    for (int i = index + 1; i < oldSize; i++) {
        newArray[i - 1] = oldArray[i];
    }
    
    // Освобождаем старый массив
    delete[] oldArray;
    
    // Возвращаем новый массив
    return newArray;
}

// Функция для удаления элемента по значению (удаляет первое вхождение)
int* removeByValue(int* oldArray, int oldSize, int value, int& newSize) {
    // Ищем элемент с таким значением
    int index = -1;
    for (int i = 0; i < oldSize; i++) {
        if (oldArray[i] == value) {
            index = i;
            break;  // нашли, выходим из цикла
        }
    }
    
    // Если не нашли
    if (index == -1) {
        cout << "Элемент со значением " << value << " не найден!" << endl;
        newSize = oldSize;
        return oldArray;
    }
    
    // Создаём новый массив на 1 элемент меньше
    newSize = oldSize - 1;
    int* newArray = new int[newSize];
    
    // Копируем элементы до найденного
    for (int i = 0; i < index; i++) {
        newArray[i] = oldArray[i];
    }
    
    // Копируем элементы после найденного (со сдвигом)
    for (int i = index + 1; i < oldSize; i++) {
        newArray[i - 1] = oldArray[i];
    }
    
    // Освобождаем старый массив
    delete[] oldArray;
    
    cout << "Элемент " << value << " удалён!" << endl;
    
    // Возвращаем новый массив
    return newArray;
}

// Функция для обработки одного массива
void processArray() {
    // Запрашиваем размер массива
    int size = inputNumber("Введите размер массива (от 1 до 100): ");
    
    // Проверяем размер
    while (size < 1 || size > 100) {
        cout << "Ошибка: размер должен быть от 1 до 100!" << endl;
        size = inputNumber("Введите размер массива: ");
    }
    
    // Создаём массив
    int* arr = new int[size];
    
    // Заполняем массив
    fillArray(arr, size);
    
    // Выводим исходный массив
    printArray(arr, size, "Исходный массив");
    
    // Цикл для удаления элементов
    while (true) {
        // Если массив пуст, выходим
        if (size == 0) {
            cout << "Массив пуст, удаление невозможно!" << endl;
            break;
        }
        
        cout << endl;
        cout << "1. Удалить элемент по номеру" << endl;
        cout << "2. Удалить элемент по значению" << endl;
        cout << "3. Завершить" << endl;
        cout << "Выберите действие: ";
        
        int choice = inputNumber("");
        
        if (choice == 3) {
            break;
        }
        
        if (choice == 1) {
            // Удаление по номеру
            int position = inputNumber("Какой элемент удалить (от 1 до " + to_string(size) + ")? ");
            int newSize;
            arr = removeElement(arr, size, position, newSize);
            size = newSize;
            if (size > 0) {
                printArray(arr, size, "Новый массив");
            }
        }
        else if (choice == 2) {
            // Удаление по значению
            int value = inputNumber("Какое значение удалить? ");
            int newSize;
            arr = removeByValue(arr, size, value, newSize);
            size = newSize;
            if (size > 0) {
                printArray(arr, size, "Новый массив");
            }
        }
        else {
            cout << "Ошибка: неверный выбор!" << endl;
        }
    }
    
    // Освобождаем память (если массив не пуст)
    if (size > 0) {
        delete[] arr;
    }
}

int main() {
    setlocale(LC_ALL, "Russian");
    
    cout << "=== Удаление элементов из массива ===" << endl;
    
    // Главный цикл программы
    while (true) {
        processArray();
        
        cout << endl;
        cout << "Хотите обработать ещё один массив? (1 - да, 0 - нет): ";
        int choice = inputNumber("");
        
        if (choice == 0) {
            cout << "До свидания!" << endl;
            break;
        }
    }
    
    return 0;
}
```
