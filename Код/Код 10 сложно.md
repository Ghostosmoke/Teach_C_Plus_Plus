## Вариант Б: Улучшенный (для хорошего ученика)

```cpp
#include <iostream>
#include <limits>
using namespace std;

// Функция для ввода размера матрицы с проверкой
int inputSize() {
    int n;
    while (true) {
        cout << "Введите размер матрицы (1-10): ";
        cin >> n;
        
        if (cin.fail()) {
            cin.clear();
            cin.ignore(numeric_limits<streamsize>::max(), '\n');
            cout << "Ошибка: введите число!" << endl;
            continue;
        }
        
        if (n >= 1 && n <= 10) {
            break;
        }
        
        cout << "Ошибка: размер должен быть от 1 до 10!" << endl;
    }
    return n;
}

// Функция для заполнения матрицы
void fillMatrix(int matrix[][10], int n) {
    cout << "Введите элементы матрицы (" << n << "x" << n << "):" << endl;
    for (int i = 0; i < n; i++) {
        cout << "Строка " << (i + 1) << ": ";
        for (int j = 0; j < n; j++) {
            cin >> matrix[i][j];
        }
    }
}

// Функция для вывода матрицы
void printMatrix(int matrix[][10], int n) {
    for (int i = 0; i < n; i++) {
        for (int j = 0; j < n; j++) {
            cout << matrix[i][j] << "\t";
        }
        cout << endl;
    }
}

// Функция для поворота матрицы на 90 градусов по часовой стрелке
void rotateMatrix(int source[][10], int result[][10], int n) {
    for (int i = 0; i < n; i++) {
        for (int j = 0; j < n; j++) {
            result[j][n - 1 - i] = source[i][j];
        }
    }
}

// Функция для обработки одной матрицы
void processMatrix() {
    int n = inputSize();
    
    int matrix[10][10];
    int result[10][10];
    
    fillMatrix(matrix, n);
    
    cout << "\nИсходная матрица:" << endl;
    printMatrix(matrix, n);
    
    rotateMatrix(matrix, result, n);
    
    cout << "\nПосле поворота на 90 градусов по часовой стрелке:" << endl;
    printMatrix(result, n);
}

int main() {
    setlocale(LC_ALL, "Russian");
    
    cout << "=== Поворот матрицы на 90 градусов ===" << endl;
    
    while (true) {
        processMatrix();
        
        cout << "\nХотите обработать ещё одну матрицу? (1 - да, 0 - нет): ";
        int choice;
        cin >> choice;
        
        if (choice == 0) {
            cout << "До свидания!" << endl;
            break;
        }
    }
    
    return 0;
}
```
