
## Вариант А: Простой (для новичков)

### main.cpp
```cpp
#include <iostream>

using namespace std;

int main() {
    // Запрашиваем у пользователя размеры матрицы
    int rows, cols;
    
    cout << "Введите количество строк: ";
    cin >> rows;
    
    cout << "Введите количество столбцов: ";
    cin >> cols;
    
    // Проверяем, что размеры правильные
    if (rows < 1 || rows > 10 || cols < 1 || cols > 10) {
        cout << "Ошибка: размеры должны быть от 1 до 10!" << endl;
        return 1;
    }
    
    // Создаём динамическую матрицу
    // Это массив указателей, каждый указатель ведёт на строку
    int** matrix = new int*[rows];  // создаём массив из rows указателей
    
    // Для каждого указателя выделяем память под строку из cols элементов
    for (int i = 0; i < rows; i++) {
        matrix[i] = new int[cols];
    }
    
    // Заполняем матрицу числами от пользователя
    cout << "Введите " << rows * cols << " чисел для матрицы:" << endl;
    for (int i = 0; i < rows; i++) {
        for (int j = 0; j < cols; j++) {
            cout << "Элемент [" << i << "][" << j << "]: ";
            cin >> matrix[i][j];
        }
    }
    
    // Выводим матрицу на экран в виде таблицы
    cout << endl << "Ваша матрица:" << endl;
    for (int i = 0; i < rows; i++) {
        for (int j = 0; j < cols; j++) {
            cout << matrix[i][j] << "\t";  // \t - это символ табуляции (отступ)
        }
        cout << endl;  // после каждой строки переходим на новую строку
    }
    
    // Находим сумму всех элементов
    int sum = 0;
    for (int i = 0; i < rows; i++) {
        for (int j = 0; j < cols; j++) {
            sum += matrix[i][j];  // добавляем каждый элемент к сумме
        }
    }
    cout << endl << "Сумма всех элементов: " << sum << endl;
    
    // Находим максимальный и минимальный элементы
    int max = matrix[0][0];  // предполагаем, что первый элемент - максимум
    int min = matrix[0][0];  // предполагаем, что первый элемент - минимум
    
    for (int i = 0; i < rows; i++) {
        for (int j = 0; j < cols; j++) {
            // Проверяем, больше ли текущий элемент текущего максимума
            if (matrix[i][j] > max) {
                max = matrix[i][j];
            }
            // Проверяем, меньше ли текущий элемент текущего минимума
            if (matrix[i][j] < min) {
                min = matrix[i][j];
            }
        }
    }
    
    cout << "Максимальный элемент: " << max << endl;
    cout << "Минимальный элемент: " << min << endl;
    
    // Считаем количество чётных и нечётных элементов
    int evenCount = 0;  // количество чётных
    int oddCount = 0;   // количество нечётных
    
    for (int i = 0; i < rows; i++) {
        for (int j = 0; j < cols; j++) {
            // Проверяем, делится ли число на 2 без остатка
            if (matrix[i][j] % 2 == 0) {
                evenCount++;  // число чётное
            }
            else {
                oddCount++;   // число нечётное
            }
        }
    }
    
    cout << "Количество чётных элементов: " << evenCount << endl;
    cout << "Количество нечётных элементов: " << oddCount << endl;
    
    // Освобождаем память!
    // Сначала удаляем каждую строку
    for (int i = 0; i < rows; i++) {
        delete[] matrix[i];
    }
    // Потом удаляем массив указателей
    delete[] matrix;
    
    return 0;
}
```
