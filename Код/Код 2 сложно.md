### main.cpp
```cpp
#include <iostream>
#include <limits>

using namespace std;

// Функция для ввода размера массива с проверкой
int inputSize() {
    int size;
    while (true) {
        cout << "Сколько чисел будет в массиве (1-100)? ";
        cin >> size;
        
        // Проверяем, что ввод корректный
        if (cin.fail()) {
            // Очистка потока при ошибке ввода
            cin.clear();
            cin.ignore(numeric_limits<streamsize>::max(), '\n');
            cout << "Ошибка: введите число!" << endl;
            continue;
        }
        
        // Проверяем диапазон
        if (size >= 1 && size <= 100) {
            break;  // всё хорошо, выходим из цикла
        }
        
        cout << "Ошибка: размер должен быть от 1 до 100!" << endl;
    }
    return size;
}

// Функция для заполнения массива
void fillArray(int* arr, int size) {
    cout << "Введите " << size << " чисел:" << endl;
    for (int i = 0; i < size; i++) {
        cout << "Число " << (i + 1) << ": ";
        cin >> arr[i];
    }
}

// Функция для вычисления суммы
int calculateSum(const int* arr, int size) {
    int sum = 0;
    for (int i = 0; i < size; i++) {
        sum += arr[i];
    }
    return sum;
}

// Функция для вычисления среднего
double calculateAverage(int sum, int size) {
    return static_cast<double>(sum) / size;
}

// Функция для поиска максимума
int findMax(const int* arr, int size) {
    int max = arr[0];
    for (int i = 1; i < size; i++) {
        if (arr[i] > max) {
            max = arr[i];
        }
    }
    return max;
}

// Функция для поиска минимума
int findMin(const int* arr, int size) {
    int min = arr[0];
    for (int i = 1; i < size; i++) {
        if (arr[i] < min) {
            min = arr[i];
        }
    }
    return min;
}

// Функция для вывода результатов
void printResults(int sum, double average, int max, int min) {
    cout << endl;
    cout << "=== Результаты ===" << endl;
    cout << "Сумма: " << sum << endl;
    cout << "Среднее: " << average << endl;
    cout << "Максимум: " << max << endl;
    cout << "Минимум: " << min << endl;
}

// Функция для обработки одного массива
void processArray() {
    // Ввод размера
    int size = inputSize();
    
    // Создание динамического массива
    int* arr = new int[size];
    
    // Заполнение массива
    fillArray(arr, size);
    
    // Вычисления
    int sum = calculateSum(arr, size);
    double average = calculateAverage(sum, size);
    int max = findMax(arr, size);
    int min = findMin(arr, size);
    
    // Вывод результатов
    printResults(sum, average, max, min);
    
    // Освобождение памяти
    delete[] arr;
}

int main() {
    setlocale(LC_ALL, "Russian");  // для корректного отображения русского текста
    
    cout << "=== Анализ массива чисел ===" << endl;
    
    // Главный цикл программы
    while (true) {
        processArray();
        
        // Спрашиваем, хочет ли пользователь продолжить
        cout << endl;
        cout << "Хотите обработать ещё один массив? (1 - да, 0 - нет): ";
        int choice;
        cin >> choice;
        
        if (choice == 0) {
            cout << "До свидания!" << endl;
            break;
        }
    }
    
    return 0;
}
```
